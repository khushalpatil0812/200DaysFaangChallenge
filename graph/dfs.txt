// DFS (Recursive)

void dfsRecursive(int vertex, boolean[] visited) {

    visited[vertex] = true;
    System.out.print(vertex + " ");

    for (int neighbor : adjList.get(vertex)) {

        if (!visited[neighbor]) {
            dfsRecursive(neighbor, visited);
        }
    }
}
